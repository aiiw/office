               row_number() over(partition by pv.sfaadocno order by pv.seq desc) rn
        
          from (select t0.sfaadocno sfaadocno,
          t0.sfaauapstdt,
                       t0.sfaa019,
                       t0.sfaa020,
                       temp.path,
                       t0.sfaa010   料号1,

                       x.xmdadocno xmdadocno,
                       x.xmda033 xmda033
                      
                      
                
                  from sfcb_t t1
                 inner join sfaa_t t0
                    on t1.sfcbdocno = t0.sfaadocno
                   and t0.sfaaent = 100
                      --  and t0.sfaasite = 'R'
                   and t0.sfaastus not in ('N', 'X')
                
                  left join xmda_t x
                    on x.xmdadocno = t0.sfaaua001
                      --  and x.xmdasite = 'R'
                   and x.xmdastus not in ('N', 'X')
                  left join imaal_t l
                    on l.imaal001 = t0.sfaa010
                   and l.imaal002 = 'zh_CN'
                   and l.imaalent = '100'
                
                  left join oocql_t l
                    on l.oocql002 = t1.sfcb003
                   and l.oocqlent = '100'
                   and l.oocql001 = '221'
                  left join (select distinct t.sfaadocno sfaadocno1,
                                            connect_by_root(t.sfaadocno) root,
                                            sys_connect_by_path(t.sfaadocno,
                                                                '/') path
                              from sfaa_t t
                             where t.sfaasite = 'W'
                               and t.sfaaent = 100
                               and t.sfaastus not in ('N', 'X')
                             start with t.sfaa021 is null
                            CONNECT BY PRIOR t.sfaadocno = t.sfaa025) temp
                    on t0.sfaadocno = temp.sfaadocno1
                
                  left join (select *
                              from (select distinct b.sfcbdocno sfcbdocno,
                                                    b.sfcb002   seq,
                                                    b.sfcb003   sfcb003,
                                                    o.oocql004  oocql004,

                                      left join sfaa_t t0
                                        on t0.sfaadocno = a.sfcadocno
                                    
                                      left join oocql_t o
                                        on b.sfcb003 = o.oocql002
                                       and o.oocql001 = '221'
                                       and o.oocql003 = 'zh_CN'
                                       and o.oocqlent = '100'
                                    
                                      left join ecaa_t g
                                        on g.ecaa001 = b.sfcb011
                                       and g.ecaaent = b.sfcbent
                                       and g.ecaasite = b.sfcbsite
                                    
                                     where 1 = 1

                                           --  where f.sfca005 = '1')
                                          -- and a.sfcadocno = 'WKS40-20080502312'
                                          --and substr(a.sfcadocno,2,4)='LS05'
                                       and a.sfca005 = '1'
                                       and b.sfcb016 = 'Y')) temp1 --temp1 只显示报工点的步骤,将
                
                    on t0.sfaadocno = temp1.sfcbdocno
                   and t1.sfcb003 = temp1.sfcb003

                              left join sfcb_t b
                                on a.sfcadocno = b.sfcbdocno

                                   (select f.sfca001
                                      from sfca_t f
                                     where f.sfca005 = '1')
                             group by a.sfcadocno) temp2 --temp2 用来关联整条工艺步骤 allname
