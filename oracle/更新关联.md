UPDATE sfab_t bb
   SET bb.sfab008 =
       (SELECT x.xmda033
          FROM sfaa_t a
          LEFT JOIN sfab_t b
            ON a.sfaadocno = b.sfabdocno
          LEFT JOIN xmdc_t c
            ON c.xmdcdocno = b.sfab002
           AND c.xmdcseq = b.sfab003
          LEFT JOIN xmda_t x
            ON x.xmdadocno = c.xmdcdocno
         WHERE a.sfaasite = 'H'
           and bb.sfabdocno = b.sfabdocno
           and bb.sfabseq = b.sfabseq)

 where bb.sfabdocno in (select a.sfaadocno
                          from sfaa_t a
                          left join sfab_t b
                            on a.sfaadocno = b.sfabdocno
                         where a.sfaasite = 'H'
                           and b.sfab008 is null)
