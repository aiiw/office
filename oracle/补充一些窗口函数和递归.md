with main as (select count(p.n1) over(partition by p.n1) countbyn1,
       count(p.n2) over(partition by p.n2) countbyn2,
       p.n1 n1,
       p.n2 n2,
       p.sfaadocno2 root,
       p.sfaadocno3 path,
       p.l l
  from (select t.sfaadocno n1,
               t.sfaa021 n2,
               level l,
                connect_by_root(t.sfaadocno) sfaadocno2,
               sys_connect_by_path(t.sfaadocno, '/') sfaadocno3
          from sfaa_t t
        where t.sfaaent=100 and t.sfaastus <>'X'
        start with t.sfaa021 is null
        CONNECT BY PRIOR t.sfaadocno = t.sfaa021) p)
        
       select * from main a where 1=1 and a.root='WKS40-20080501387'
