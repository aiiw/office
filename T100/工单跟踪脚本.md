```sql
select distinct t0.sfaadocno   单据,
                x.xmdadocno    订单号,
                x.xmda033      生产单号,
                t0.sfaauapstdt 过账时间,
                t0.sfaa019     开工时间,
                t0.sfaa020     完工时间,
                temp.path      路径,
                t0.sfaa010     料号1,
                l.imaal003     品名,
                t0.sfaa012     生产数量,
                --temp2.names  全工艺步骤,
                --temp2.names1 全工作站,
                temp3.当前步骤,
                temp3.第几步,
                temp3.总步数,
                temp3.工艺比率,
                
                ptemp1.*

  from sfcb_t t1
 inner join sfaa_t t0
    on t1.sfcbdocno = t0.sfaadocno
   and t0.sfaaent = 100
      --  and t0.sfaasite = 'R'
   and t0.sfaastus not in ('N', 'X')

  left join xmda_t x
    on x.xmdadocno = t0.sfaaua001
      --  and x.xmdasite = 'R'
   and x.xmdastus not in ('N', 'X')
  left join imaal_t l
    on l.imaal001 = t0.sfaa010
   and l.imaal002 = 'zh_CN'
   and l.imaalent = '100'

  left join oocql_t l
    on l.oocql002 = t1.sfcb003
   and l.oocqlent = '100'
   and l.oocql001 = '221'
  left join (select distinct t.sfaadocno sfaadocno1,
                             connect_by_root(t.sfaadocno) root,
                             sys_connect_by_path(t.sfaadocno, '/') path
               from sfaa_t t
              where t.sfaasite = 'W'
                and t.sfaaent = 100
                and t.sfaastus not in ('N', 'X')
              start with t.sfaa021 is null
             CONNECT BY PRIOR t.sfaadocno = t.sfaa025) temp
    on t0.sfaadocno = temp.sfaadocno1

  left join (select distinct temp1.sfcbdocno,
                             
                             SUM(CASE
                                   WHEN temp1.ecaa002 = '装配站' THEN
                                    NVL(temp1.num1, 0)
                                   ELSE
                                    0
                                 END) AS "装配站",
                             SUM(CASE
                                   WHEN temp1.ecaa002 = '砂光站' THEN
                                    NVL(temp1.num1, 0)
                                   ELSE
                                    0
                                 END) AS "砂光站",
                             SUM(CASE
                                   WHEN temp1.ecaa002 = '抛光站' THEN
                                    NVL(temp1.num1, 0)
                                   ELSE
                                    0
                                 END) AS "抛光站",
                             SUM(CASE
                                   WHEN temp1.ecaa002 = '复底站' THEN
                                    NVL(temp1.num1, 0)
                                   ELSE
                                    0
                                 END) AS "复底站",
                             SUM(CASE
                                   WHEN temp1.ecaa002 = '拉伸站' THEN
                                    NVL(temp1.num1, 0)
                                   ELSE
                                    0
                                 END) AS "拉伸站",
                             SUM(CASE
                                   WHEN temp1.ecaa002 = '开料站' THEN
                                    NVL(temp1.num1, 0)
                                   ELSE
                                    0
                                 END) AS "开料站"
             
               from (select distinct b.sfcbdocno sfcbdocno,
                                     b.sfcb002 seq,
                                     b.sfcb003 sfcb003,
                                     b.sfcb033 num1,
                                     o.oocql004 oocql004,
                                     b.sfcb011 sfcb011,
                                     g.ecaa002 ecaa002,
                                     row_number() over(partition by b.sfcbdocno, g.ecaa002 order by b.sfcb002 desc) rn
                       from sfca_t a
                       left join sfcb_t b
                         on a.sfcadocno = b.sfcbdocno
                       left join sfaa_t t0
                         on t0.sfaadocno = a.sfcadocno
                     
                       left join oocql_t o
                         on b.sfcb003 = o.oocql002
                        and o.oocql001 = '221'
                        and o.oocql003 = 'zh_CN'
                        and o.oocqlent = '100'
                     
                       left join ecaa_t g
                         on g.ecaa001 = b.sfcb011
                        and g.ecaaent = b.sfcbent
                        and g.ecaasite = b.sfcbsite
                     
                      where 1 = 1
                           
                           --and a.sfcadocno = 'WKS40-20080502312'
                           --and substr(a.sfcadocno,2,4)='LS05'
                        and a.sfca005 = '1' --and b.sfcb016 = 'Y' 暂时不加  去掉报工
                        and b.sfcb033 <> 0) temp1
              where temp1.rn = 1
              group by temp1.sfcbdocno) ptemp1

    on t0.sfaadocno = ptemp1.sfcbdocno

  left join (select a.sfcadocno,
                    listagg(o.oocql004, '->') WITHIN GROUP(ORDER BY b.sfcb002) names,
                    listagg(g.ecaa002, '->') WITHIN GROUP(ORDER BY b.sfcb002) names1
               from sfca_t a
               left join sfcb_t b
                 on a.sfcadocno = b.sfcbdocno
             
               left join oocql_t o
                 on b.sfcb003 = o.oocql002
                and o.oocql001 = '221'
                and o.oocql003 = 'zh_CN'
                and o.oocqlent = '100'
             
               left join ecaa_t g
                 on g.ecaa001 = b.sfcb011
                and g.ecaaent = b.sfcbent
                and g.ecaasite = b.sfcbsite
              where 1 = 1
                and b.sfcb001 in
                    (select f.sfca001 from sfca_t f where f.sfca005 = '1')
              group by a.sfcadocno) temp2 --temp2 用来关联整条工艺步骤 allname
    on t0.sfaadocno = temp2.sfcadocno

  left join (select pv.sfcadocno,
                    max(case
                          when pv.sfcb033 <> 0 then
                           pv.oocql004
                          else
                           '0'
                        end) 当前步骤,
                    sum(case
                          when pv.sfcb033 <> 0 then
                           1
                          else
                           0
                        end) 第几步,
                    max(pv.total1) 总步数,
                    sum(case
                          when pv.sfcb033 <> 0 then
                           1
                          else
                           0
                        end) / max(pv.total1) 工艺比率
             
               from (select a.sfcadocno,
                            b.sfcb033,
                            b.sfcb002,
                            o.oocql004,
                            -- row_number() over(partition by a.sfcadocno order by b.sfcb002 desc),
                            --  first_value(o.oocql004) over(partition by a.sfcadocno order by b.sfcb002 desc),
                            
                            count(a.sfcadocno) over(partition by a.sfcadocno) total1
                     
                       from sfca_t a
                       left join sfcb_t b
                         on a.sfcadocno = b.sfcbdocno
                     
                       left join oocql_t o
                         on b.sfcb003 = o.oocql002
                        and o.oocql001 = '221'
                        and o.oocql003 = 'zh_CN'
                        and o.oocqlent = '100'
                     
                       left join ecaa_t g
                         on g.ecaa001 = b.sfcb011
                        and g.ecaaent = b.sfcbent
                        and g.ecaasite = b.sfcbsite
                      where 1 = 1
                        and b.sfcb001 in
                            (select f.sfca001
                               from sfca_t f
                              where f.sfca005 = '1')) pv
              group by pv.sfcadocno
             -- and b.sfcb033 <> 0
             ) temp3
    on t0.sfaadocno = temp3.sfcadocno

--left join () temp2
 where 1 = 1
   and t0.sfaastus <> 'X' and t0.sfaa057='1' --1为自制,2 为委外
   and t1.sfcb001 in (select f.sfca001 from sfca_t f where f.sfca005 = '1')
   and temp.root = 'WMS05-24070400050'
   or t0.sfaauapstdt>sysdate-30
 order by t0.sfaadocno
```
