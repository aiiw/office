  pv.opname 全工作站,
  pv.num1 报工数量,
  pv.xmdadocno 订单号,
  pv.xmda033 生产单号,
               row_number() over(partition by pv.sfaadocno,pv.ecaa002 order by pv.seq desc) rn
        
          from (select t0.sfaadocno sfaadocno,
          t0.sfaauapstdt,
                       t0.sfaa019 sfaa019,
                       t0.sfaa020 sfaa020,
                       temp.path path,
                       t0.sfaa010   料号1,

                       x.xmda033 xmda033

                              from (select distinct b.sfcbdocno sfcbdocno,
                                                    b.sfcb002   seq,

                                                    o.oocql004  oocql004,

                   and t1.sfcb003 = temp1.sfcb003

                                   listagg(g.ecaa002, '->') WITHIN GROUP(ORDER BY b.sfcb002) names1
